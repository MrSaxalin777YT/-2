    <script>
      const tg = window.Telegram.WebApp;
      tg.ready();

      const canvas = document.getElementById('wheelCanvas');
      const ctx = canvas.getContext('2d');
      const spinBtn = document.getElementById('spinBtn');
      const resultBox = document.getElementById('resultBox');
      const resultText = document.getElementById('resultText');
      const resultDesc = document.getElementById('resultDesc');

      // Призы и цвета секторов
      const prizes = [
        { prize: 'Деревянная кирка', desc: 'Базовый инструмент', color: '#8B5E3C' },
        { prize: 'Железный слиток', desc: 'Для крафта брони', color: '#5e6b7a' },
        { prize: 'Семена пшеницы', desc: 'Для фермы', color: '#7F9A55' },
        { prize: 'Зелье здоровья', desc: '+50 HP', color: '#c12f2f' },
        { prize: 'Карта сокровищ', desc: 'Указывает на лут', color: '#f1c40f' },
        { prize: 'Алмазный блок', desc: 'Редкий ресурс', color: '#3498db' },
        { prize: 'Набор еды', desc: 'Сытость +30%', color: '#e67e22' },
        { prize: 'Чертеж базы', desc: 'Схема постройки', color: '#9b59b6' }
      ];

      let isSpinning = false;

      function resize() {
        const size = Math.min(window.innerWidth, 320);
        canvas.width = size;
        canvas.height = size;
        drawWheel();
      }
      window.addEventListener('resize', resize);
      resize();

      function drawWheel() {
        const sectors = prizes.length;
        const radius = canvas.width / 2;
        const angle = (2 * Math.PI) / sectors;

        for (let i = 0; i < sectors; i++) {
          ctx.beginPath();
          ctx.fillStyle = prizes[i].color;
          ctx.moveTo(radius, radius);
          ctx.arc(radius, radius, radius, i * angle, (i + 1) * angle);
          ctx.lineTo(radius, radius);
          ctx.fill();

          // Текст приза (упрощённо — можно сделать красивее)
          ctx.save();
          ctx.translate(radius, radius);
          ctx.rotate(i * angle + angle / 2);
          ctx.textAlign = 'right';
          ctx.fillStyle = '#fff';
          ctx.font = '12px Arial';
          ctx.fillText(prizes[i].prize, radius - 12, 6);
          ctx.restore();
        }
      }

      spinBtn.addEventListener('click', () => {
        if (isSpinning) return;
        isSpinning = true;
        spinBtn.disabled = true;
        resultBox.style.display = 'none';

        // Случайный сектор
        const randomIndex = Math.floor(Math.random() * prizes.length);
        const prize = prizes[randomIndex];

        // Длительность и угол вращения (добавим случайный оборот)
        const spins = 6 + Math.random() * 4; // 6–10 оборотов
        const degreesPerSpin = 360;
        const totalDegrees = spins * degreesPerSpin + (randomIndex * (360 / prizes.length));
        const duration = 4000 + Math.random() * 2000; // 4–6 секунд

        const startTime = performance.now();

        function animate(currentTime) {
          const elapsed = currentTime - startTime;
          const progress = Math.min(elapsed / duration, 1);

          // Easing (ease-out)
          const ease = 1 - Math.pow(1 - progress, 3);

          const currentDegrees = totalDegrees * ease;
          canvas.style.transform = `rotate(${currentDegrees}deg)`;

          if (progress < 1) {
            requestAnimationFrame(animate);
          } else {
            // Конец вращения
            isSpinning = false;
            spinBtn.disabled = false;
            resultText.textContent = prize.prize;
            resultDesc.textContent = prize.desc;
            resultBox.style.display = 'block';

            // Отправляем результат в бота
            const payload = JSON.stringify({
              action: 'wheel_result',
              prize_id: randomIndex,
              prize: prize.prize,
              desc: prize.desc
            });
            tg.sendData(payload);
          }
        }
        requestAnimationFrame(animate);
      });
